  /** Contains the formats available to Gias. */
  private readonly formats = {
    "it": "dd/MM/yyyy",
    "en": "MM/dd/yyyy",
    "fr": "dd/MM/yyyy",
    "pt": "dd-MM-yyyy",
  }

  constructor(private transloco: TranslocoService) { }

  /** Returns the format used to visualize dates.
   * @param locale the language to use to retrive the date format. If not specified, the active language is used as default.
   * @returns a string indicating the date format
   */
  public getDateFormat(locale?: string): string {
    if (!locale) {
      locale = this.transloco.getActiveLang();
    }
    return this.formats[locale];
  }

  /** Parses a locale date string to a Date object.
   * @param dateString the locale date string to parse (i.e. "26/03/2024"). Note: this string must not specify the time.
   * @param locale the language to use to retrive the date format. If not specified, the active language is used as default.
   * @returns the date indicated by `dateString`
   */
  public parseDate(dateString: string, locale?: string): Date {
    if (!locale) {
      locale = this.transloco.getActiveLang();
    }
    const dateFormatParts = this.getFormatParts(locale);
    const dateParts = dateString.split(this.getSeparator(dateString));
    const options = { "dd" : 0, "MM" : 0, "yyyy": 0};
    dateFormatParts.forEach((key, i) => options[key] = +dateParts[i]);
    options["MM"] = options["MM"] - 1;
    return new Date(options["yyyy"], options["MM"], options["dd"]);
  }

  private getFormatParts(locale?: string): string[] {
    const format = this.getDateFormat(locale);
    return format.split(this.getSeparator(format));
  }

  private getSeparator(dateFormat: string): string {
    if (dateFormat.includes('/')) return '/';
    else if (dateFormat.includes('-')) return '-';
    else if (dateFormat.includes('. ')) return '. ';
    else if (dateFormat.includes('.')) return '.';
    else throw new Error("Format separator not handled");
  }
}
