export const loader = LOCALIZATION_LANGUAGES.reduce((acc, lang) => {
  acc[lang] = () => import(`./i18n/${lang}.json`);
  return acc;
}, {});

@NgModule({
  imports: [

    ZooComponent
  ],

      useValue: {
        scope: 'zoo',
        loader,
        multi: true
      }
    },
    {provide: LOADING_TOKEN, useClass: LoadingService}
  ]
})
export class ZooModule {
