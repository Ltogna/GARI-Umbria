@Component({
    standalone: false,
    selector: 'app-grid-impianti',
    templateUrl: './grid-impianti.component.html',
    styleUrls: ['./grid-impianti.component.css'],
    providers: [
        ...generateGridProviders(GridImpiantiHttpService, GridImpiantiComponent), CalcoloSuperficiService
    ]
})
export class GridImpiantiComponent implements OnInit, AfterViewInit {

    @ViewChild(TooltipDirective) public tooltipDir: TooltipDirective;
    @ViewChild('GridImpianti') GridImpiantiElRef: GiasKendoGridComponent;

    private data: GridImpiantiServerResult | null = null;
    private dataLoaded$ = new Subject<boolean>();
    private viewLoaded$ = new Subject<boolean>();
    private rowsLoaded$ = new Subject<boolean>();

    constructor(
        private gridimpiantiservice: GridImpiantiService,
        private qdcservice: QdCService,
        private gridpublicService: GridPublicService,
        private gisGeometrySelectionService: GISGeometrySelectionService,
        @Inject(GRID_HTTP_TOKEN) private gridhttpService: GridImpiantiHttpService
    ) {
        this.gridhttpService.impiantiLoaded$.subscribe(data => {
            this.data = data;
            this.dataLoaded$.next(true);
        });

        this.gisGeometrySelectionService.geometrySelectedFromMap$
            .subscribe((keys) => this.handleSelectedGeometries(keys));

        this.qdcservice.eventiPostSelectionChangeGridImpianti
            .subscribe(rows => {
                if (rows) {
                    this.gridimpiantiservice.EventiPostselectionChangeGridImpianti(rows);
                }
            });

        this.rowsLoaded$
            .asObservable()
            .pipe(take(1))
            .subscribe(() => this.handleSelectedGeometries(this.gisGeometrySelectionService.getCurrentSelection()));
    }

    ngOnInit(): void {
        this.qdcservice.GridImpiantiHttpService = this.gridhttpService;
        this.qdcservice.GridImpiantiPublicService = this.gridpublicService;
    }

    ngAfterViewInit(): void {
        this.gridimpiantiservice.settooltip(this.tooltipDir);
        this.viewLoaded$.next(true);
    }

    onSelected(event: SelectionEvent): void {
        const selected = event.selectedRows.map((item) => this.getKeyFromSelectedItem(item.dataItem));
        this.gisGeometrySelectionService.addFromOutside(...selected as string[]);

        const deselected = event.deselectedRows.map((item) => this.getKeyFromSelectedItem(item.dataItem));
        this.gisGeometrySelectionService.removeFromOutside(...deselected as string[]);
    }

    rowsLoaded() {
        this.rowsLoaded$.next(true);

        if(this.qdcservice.Apertura_QdC && this.qdcservice.Apertura_QdC.flag_QdC_Aperta_da_Altra_Pagina && this.qdcservice.Apertura_QdC.flag_Selezione_Impianti){

            if(this.qdcservice.GridImpiantiPublicService && this.qdcservice.GridImpiantiPublicService?.getValue()?.data?.rows?.length > 0){

                let righe_selezionate_grid = this.gridimpiantiservice.kGetElementiSelezionati(cloneDeep(this.qdcservice.GridImpiantiPublicService.getValue().data.rows));

                if(righe_selezionate_grid && righe_selezionate_grid.length > 0){

                  this.qdcservice.Apertura_QdC.flag_Selezione_Impianti = false;

                  this.qdcservice.AggiornaRigheSelezionateGriglia(righe_selezionate_grid,[]);

                  this.gridimpiantiservice.EventiPostselectionChangeGridImpianti(this.qdcservice.GridImpiantiPublicService.getValue().data.rows,true);
                }

            }


        }
    }

    private handleSelectedGeometries(keys: [string, boolean][]): void {
        if (this.data == null || this.GridImpiantiElRef.rows == null) {
            return;
        }

        // if service is not initialized, get present selection from grid
        if (!this.gisGeometrySelectionService.isInitialized()) {
            const preSelected = this.GridImpiantiElRef.rows
                .filter((row) => row['Selected'] === true)
                .map((row) => this.getKeyFromSelectedItem(row as GridRowElement));

            if (preSelected.length > 0) {
                this.gisGeometrySelectionService.addFromOutside(...preSelected as string[]);
            }

            return;
        }

        const nonSelected: { key: string; value: GridRowElement }[] = [];

        for (const row of this.GridImpiantiElRef.rows) {
            const rowKey = this.getKeyFromSelectedItem(row as GridRowElement);

            const key = keys.find((item) => item[0].includes(rowKey));
            if (key == null) {
                nonSelected.push({ key: rowKey, value: row as GridRowElement });
                continue;
            }

            const isSelected = key[1];
            row['Selected'] = isSelected;

        }

        // this.logNonSelected(nonSelected);
        let selectedRows = this.GridImpiantiElRef.rows.filter(row =>(row['Selected'] === true));
        let deselectedRows = this.GridImpiantiElRef.rows.filter(row =>(row['Selected'] === false));
        this.qdcservice.AggiornaRigheSelezionateGriglia(selectedRows, deselectedRows);
        this.GridImpiantiElRef.refresh();
    }

    private getKeyFromSelectedItem(item: GridRowElement): string {
        return `§${item.PIVA}§${item.SA_COD}§${item.Campo_Cod}§${item.APPEZZA}§${item.ID_REG}`;
    }

    private logNonSelected(nonSelected: { key: string; value: GridRowElement }[]): void {
        console.log('The following rows have not been selected:');
        console.table(nonSelected.map(item => ({ 'name': item.value.APP_NOME, 'key': item.key })));

        console.log('Table of current selection:');
        console.table(this.gisGeometrySelectionService.getCurrentSelection());
    }
}
